<script>
  // 每次调用都会重播进入 → 停留 → 退出，总时长 3.5 秒。
  function showToast(message) {
    const toast = document.querySelector('.jjb-toast');
    toast.querySelector('.jjb-toast__copy').textContent = message;
    toast.classList.remove('is-visible');
    void toast.offsetWidth;
    toast.classList.add('is-visible');
  }
</script>
